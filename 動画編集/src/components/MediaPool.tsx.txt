import React, { useRef } from 'react';
import { useEditorStore } from '../store';

export const MediaPool: React.FC = () => {
  const { mediaFiles, addMediaFile, setCurrentVideoUrl, currentVideoUrl, addClipToTimeline, addTextClipToTimeline } = useEditorStore();
  const fileInputRef = useRef<HTMLInputElement>(null);

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.files && e.target.files.length > 0) {
      Array.from(e.target.files).forEach((file) => addMediaFile(file));
    }
    if (fileInputRef.current) fileInputRef.current.value = '';
  };

  return (
    <section className="w-80 border-r border-editor-border bg-editor-panel flex flex-col">
      <div className="p-3 border-b border-editor-border flex items-center justify-between">
        <h2 className="text-xs font-bold uppercase tracking-wider text-gray-400">素材フォルダ</h2>
        <div className="flex space-x-1">
          <button 
            onClick={() => addTextClipToTimeline()}
            className="text-xs bg-gray-700 hover:bg-gray-600 text-gray-200 px-2 py-1 rounded transition-colors"
          >
            T テロップ追加
          </button>
          <button 
            onClick={() => fileInputRef.current?.click()}
            className="text-xs bg-editor-accent hover:bg-editor-accentHover text-white px-2 py-1 rounded transition-colors"
          >
            + ファイル
          </button>
        </div>
        <input 
          type="file" 
          ref={fileInputRef} 
          onChange={handleFileChange} 
          accept="video/*,audio/*" 
          multiple 
          className="hidden" 
        />
      </div>

      <div className="flex-1 p-3 overflow-y-auto space-y-2 bg-editor-bg/30">
        {mediaFiles.length === 0 ? (
          <div className="text-xs text-gray-500 text-center pt-8 italic">
            ファイルを追加するか、上のボタンからテロップを配置してください
          </div>
        ) : (
          mediaFiles.map((file) => (
            <div 
              key={file.id}
              onClick={() => file.type === 'video' && setCurrentVideoUrl(file.url)}
              className={`border p-2 rounded cursor-pointer transition-colors relative group ${
                currentVideoUrl === file.url 
                  ? 'border-editor-accent bg-editor-accent/10' 
                  : 'border-editor-border bg-editor-panel hover:border-gray-500'
              }`}
            >
              <div className="aspect-video bg-gray-900 rounded flex items-center justify-center text-xs text-gray-500 mb-1">
                {file.type === 'video' ? '🎬 ビデオ' : '🎵 オーディオ'}
              </div>
              <div className="flex items-center justify-between">
                <p className="text-xs truncate text-gray-300 w-3/4">{file.name}</p>
                <button 
                  onClick={(e) => {
                    e.stopPropagation();
                    addClipToTimeline(file);
                  }}
                  className="text-[10px] bg-editor-border hover:bg-editor-accent px-1.5 py-0.5 rounded text-white opacity-0 group-hover:opacity-100 transition-opacity"
                >
                  ＋配置
                </button>
              </div>
            </div>
          ))
        )}
      </div>
    </section>
  );
};
