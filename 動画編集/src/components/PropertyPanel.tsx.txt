import React from 'react';
import { useEditorStore } from '../store';

export const PropertyPanel: React.FC = () => {
  const { tracks, selectedClipId, updateClipProperties } = useEditorStore();

  let selectedClip = null;
  let trackType: 'video' | 'audio' | 'text' | null = null;

  for (const track of tracks) {
    const found = track.clips.find((c) => c.id === selectedClipId);
    if (found) {
      selectedClip = found;
      trackType = track.type;
      break;
    }
  }

  if (!selectedClip) {
    return (
      <section className="w-72 border-l border-editor-border bg-editor-panel flex flex-col">
        <div className="p-3 border-b border-editor-border">
          <h2 className="text-xs font-bold uppercase tracking-wider text-gray-400">変形・エフェクト</h2>
        </div>
        <div className="flex-1 flex items-center justify-center p-4 text-center text-xs text-gray-500 italic">
          タイムライン上のクリップを選択すると、ここにプロパティが表示されます。
        </div>
      </section>
    );
  }

  return (
    <section className="w-72 border-l border-editor-border bg-editor-panel flex flex-col text-xs text-gray-200">
      <div className="p-3 border-b border-editor-border flex items-center justify-between bg-editor-stripe">
        <h2 className="font-bold uppercase tracking-wider text-editor-accent">
          {trackType === 'text' ? '📝 テキストプロパティ' : '🎬 クリッププロパティ'}
        </h2>
        <span className="text-[10px] text-gray-500 bg-editor-bg px-1.5 py-0.5 rounded border border-editor-border">
          選択中
        </span>
      </div>

      <div className="flex-1 p-4 space-y-5 overflow-y-auto">
        {trackType === 'text' && (
          <div className="space-y-2">
            <label className="text-gray-400 font-medium block">テロップ文章</label>
            <input
              type="text"
              value={selectedClip.text ?? ''}
              onChange={(e) => updateClipProperties(selectedClip.id, { text: e.target.value })}
              className="w-full bg-editor-bg border border-editor-border rounded px-2 py-1.5 text-gray-100 focus:outline-none focus:border-editor-accent"
              placeholder="表示する文字列"
            />
          </div>
        )}

        <div className="space-y-2">
          <label className="text-gray-400 font-medium block">表示時間 (秒)</label>
          <input
            type="number"
            min="1"
            max="60"
            value={selectedClip.duration}
            onChange={(e) => updateClipProperties(selectedClip.id, { duration: Math.max(1, Number(e.target.value)) })}
            className="w-24 bg-editor-bg border border-editor-border rounded px-2 py-1 text-gray-100 focus:outline-none focus:border-editor-accent"
          />
        </div>

        {trackType === 'text' && (
          <div className="space-y-4 pt-4 border-t border-editor-border/60">
            <h3 className="font-semibold text-gray-400">アニメーション (不透明度フェード)</h3>
            
            <div className="space-y-1">
              <div className="flex justify-between text-gray-400">
                <span>フェードイン</span>
                <span className="font-mono text-editor-accent">{selectedClip.fadeIn ?? 0}s</span>
              </div>
              <input
                type="range"
                min="0"
                max="3"
                step="0.1"
                value={selectedClip.fadeIn ?? 0}
                onChange={(e) => updateClipProperties(selectedClip.id, { fadeIn: Number(e.target.value) })}
                className="w-full accent-editor-accent"
              />
            </div>

            <div className="space-y-1">
              <div className="flex justify-between text-gray-400">
                <span>フェードアウト</span>
                <span className="font-mono text-editor-accent">{selectedClip.fadeOut ?? 0}s</span>
              </div>
              <input
                type="range"
                min="0"
                max="3"
                step="0.1"
                value={selectedClip.fadeOut ?? 0}
                onChange={(e) => updateClipProperties(selectedClip.id, { fadeOut: Number(e.target.value) })}
                className="w-full accent-editor-accent"
              />
            </div>
          </div>
        )}
      </div>
    </section>
  );
};
