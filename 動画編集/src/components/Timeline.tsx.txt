import React, { useRef, useState } from 'react';
import { useEditorStore } from '../store';

const PIXELS_PER_SECOND = 20;

export const Timeline: React.FC = () => {
  const { tracks, currentTime, setCurrentTime, selectedClipId, setSelectedClipId } = useEditorStore();
  const timelineRef = useRef<HTMLDivElement>(null);
  const [isScrubbing, setIsScrubbing] = useState(false);

  const calculateTimeFromMouseEvent = (e: React.MouseEvent | MouseEvent) => {
    if (!timelineRef.current) return 0;
    const rect = timelineRef.current.getBoundingClientRect();
    const offsetX = e.clientX - rect.left;
    return Math.max(0, offsetX / PIXELS_PER_SECOND);
  };

  const handlePointerDown = (e: React.MouseEvent) => {
    if ((e.target as HTMLElement).classList.contains('track-lane') || (e.target as HTMLElement).classList.contains('ruler-lane')) {
      setSelectedClipId(null);
    }
    setIsScrubbing(true);
    setCurrentTime(calculateTimeFromMouseEvent(e));
  };

  const handlePointerMove = (e: React.MouseEvent) => {
    if (isScrubbing) {
      setCurrentTime(calculateTimeFromMouseEvent(e));
    }
  };

  const handlePointerUp = () => {
    setIsScrubbing(false);
  };

  const formatTime = (time: number) => {
    const mins = Math.floor(time / 60).toString().padStart(2, '0');
    const secs = Math.floor(time % 60).toString().padStart(2, '0');
    const frames = Math.floor((time % 1) * 30).toString().padStart(2, '0');
    return `${mins}:${secs}:${frames}`;
  };

  return (
    <section className="h-64 bg-editor-panel flex flex-col overflow-hidden select-none">
      <div className="h-9 border-b border-editor-border flex items-center justify-between px-4 text-xs bg-editor-stripe">
        <div className="flex items-center space-x-3 text-gray-400">
          <button className="hover:text-editor-accent" onClick={() => setSelectedClipId(null)}>選択解除</button>
        </div>
        <div className="text-gray-400">
          時間軸: <span className="text-editor-accent font-mono">{formatTime(currentTime)}</span>
        </div>
      </div>

      <div 
        className="flex-1 overflow-x-auto overflow-y-auto relative bg-editor-bg/50"
        onMouseMove={handlePointerMove}
        onMouseUp={handlePointerUp}
        onMouseLeave={handlePointerUp}
      >
        <div ref={timelineRef} className="relative min-w-[2000px] h-full">
          
          {/* ルーラー */}
          <div 
            className="h-6 border-b border-editor-border/40 bg-editor-panel/50 cursor-text sticky top-0 z-20 ruler-lane"
            onMouseDown={handlePointerDown}
          >
            {Array.from({ length: 50 }).map((_, i) => (
              <div 
                key={i} 
                className="absolute border-l border-gray-600 text-[9px] text-gray-500 pl-1 pointer-events-none"
                style={{ left: `${i * 10 * PIXELS_PER_SECOND}px`, height: '100%' }}
              >
                00:{i * 10}
              </div>
            ))}
          </div>

          {/* トラックとクリップ */}
          <div className="p-2 space-y-2 relative z-10">
            {tracks.map((track) => (
              <div 
                key={track.id} 
                className="h-12 bg-editor-stripe border border-editor-border rounded flex items-center relative track-lane"
                onMouseDown={(e) => {
                  if (e.target === e.currentTarget) handlePointerDown(e);
                }}
              >
                <span className="text-[10px] text-gray-500 absolute left-2 top-1 z-10 bg-editor-stripe/80 px-1 rounded pointer-events-none">
                  {track.type === 'text' ? 'Text 1' : track.type === 'video' ? 'Video 1' : 'Audio 1'}
                </span>
                
                {track.clips.map((clip) => {
                  const isSelected = clip.id === selectedClipId;
                  
                  let colorClass = "bg-editor-accent/20 border-editor-accent text-editor-accent"; 
                  if (track.type === 'audio') colorClass = "bg-emerald-500/10 border-emerald-500/40 text-emerald-400";
                  if (track.type === 'text') colorClass = "bg-amber-500/20 border-amber-500 text-amber-400";

                  return (
                    <div
                      key={clip.id}
                      onClick={(e) => {
                        e.stopPropagation(); 
                        setSelectedClipId(clip.id);
                      }}
                      className={`absolute h-8 rounded flex items-center px-2 text-xs overflow-hidden cursor-pointer border ${colorClass} ${
                        isSelected ? 'ring-2 ring-white border-transparent z-20 shadow-lg' : 'hover:brightness-110'
                      }`}
                      style={{
                        left: `${clip.start * PIXELS_PER_SECOND}px`,
                        width: `${clip.duration * PIXELS_PER_SECOND}px`,
                        top: '8px'
                      }}
                    >
                      <span className="truncate font-medium">
                        {track.type === 'text' ? `📝 "${clip.text}"` : clip.name}
                      </span>
                    </div>
                  );
                })}
              </div>
            ))}
          </div>

          {/* 再生ヘッド */}
          <div 
            className="absolute top-0 bottom-0 w-px bg-red-500 pointer-events-none z-30 shadow-[0_0_4px_rgba(239,68,68,0.8)]"
            style={{ left: `${currentTime * PIXELS_PER_SECOND}px` }}
          >
            <div className="w-0 h-0 border-l-[6px] border-l-transparent border-r-[6px] border-r-transparent border-t-[8px] border-t-red-500 absolute -top-0 -left-[5.5px]"></div>
          </div>

        </div>
      </div>
    </section>
  );
};
